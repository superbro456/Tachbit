Ядер мало, программ много. Планировщик даёт каждой задаче немного времени (квант), потом ставит её в конец очереди и берёт следующую. Так все двигаются вперёд, и никто не захватывает процессор навсегда. Это вариант «по кругу» (round-robin); настоящие планировщики добавляют приоритеты, но основа та же.

Паттерн «По кругу»

Очередь задач. Каждой — квант. Не закончила — в конец очереди.

ПриложениеСервер раздаёт по чуть-чуть всем клиентам, чтобы один тяжёлый запрос не тормозил остальных.
ИграТяжёлые дела (поиск пути для 100 врагов) делят на куски: по 10 врагов за кадр.
ОСТаймер ядра Omnium тикает и по очереди переключает задачи на всех ядрах.

Код

day28.tb
// день 28: планировщик — всем по очереди понемногу
obj Task
    name: text
    left: int      // сколько ещё работы

func main()
    queue = list Task
    queue.add(Task(name: "музыка", left: 3))
    queue.add(Task(name: "браузер", left: 5))
    queue.add(Task(name: "часы", left: 1))
    slice = 2                               // квант: сколько даём за раз
    tick = 0
    repeat while len(queue) > 0
        t = queue[0]
        rest = list Task
        count i from 1 to len(queue) - 1
            rest.add(queue[i])
        work = min(slice, t.left)
        t.left -= work
        tick += work
        print("такт", tick, ":", t.name, "сделал", work)
        if t.left > 0
            rest.add(t)                     // не закончил — в конец очереди
        else
            print("   ", t.name, "готов")
        queue = rest
