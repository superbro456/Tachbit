Драйвер клавиатуры не может просить новую память на каждое нажатие — ему нужна заранее выделенная полка. Кольцевой буфер — это массив, где запись и чтение идут по кругу: дошли до конца — начинаем сначала.

Два указателя: tail — куда писать, head — откуда читать. Если полка полна, решают заранее: терять новое или перезаписывать старое.

Паттерн «Кольцо»

Фиксированный массив, два указателя, всё по модулю длины.

ПриложениеПоследние 100 строк журнала, история чата в памяти, буфер записи звука.
ИграЗапись последних 5 секунд для повтора гола, сглаживание мыши по последним кадрам.
ОСБуфер клавиатуры, сетевой карты, звуковой карты: железо пишет, драйвер читает.

Код

day27.tb
// день 27: кольцевой буфер — как драйвер копит нажатия
obj Ring
    data: list int
    head: int      // откуда читать
    tail: int      // куда писать
    used: int

func put(r: Ring, x: int) gives ok: bool
    ok = r.used < len(r.data)
    if not ok
        return                             // полно: новое теряем, старое не портим
    r.data[r.tail] = x
    r.tail = (r.tail + 1) % len(r.data)
    r.used += 1

func take(r: Ring) gives x: int
    x = r.data[r.head]
    r.head = (r.head + 1) % len(r.data)
    r.used -= 1

func main()
    r = Ring(data: [0, 0, 0, 0])
    each key in [65, 66, 67, 68, 69]       // 5 нажатий в буфер на 4
        if not put(r, key)
            print("буфер полон, потеряно:", char(key))
    repeat while r.used > 0
        print("прочитал:", char(take(r)))
